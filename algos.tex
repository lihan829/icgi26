	\begin{minipage}[t]{0.6\textwidth}
		
		\SetKwInput{KwData}{Input}
		\SetKwInput{KwResult}{Output}
		\textbf{Algorithm 1: BUFIA-AC}
		
		\KwData{$D= \{(I_1,O_1),\dots,(I_n,O_n)\}, h_0, k$
		}
		\KwResult{An environment \(c\) or \(\bot\)
		}
		
		%	
		%	\KwOut{
			%		minimal structural environment $c$
			%	}
		\(D^+ \gets \{\, I_i \mid I_i \neq O_i \,\}\)\;
		
		\(D^- \gets \{\, I_i \mid I_i = O_i \,\}\)\;
		
		$P \gets \emptyset$\;
		%	\tcp*{pruned set}
		
		$Q \gets h_0$\;
		
		\While{$Q \neq \emptyset$}{
			
			$h \gets Q.\mathrm{Dequeue}()$\;
			
			\If{$\forall I_i \in D^+\; [\textsc{match}(h,I_i,O_i,h_0)]$}{
				
				\If{$\forall I_i \in D^-\; [\neg \textsc{match}(h,I_i,O_i,h_0)]$}{
					
					\Return{$h$}\;
				}
				
				\Else{
					\ForEach{$s \in \textsc{NextSupFact}(h)$}{
						\If{$|s| < k \land (\forall p \in P)[s \not\sqsubseteq p]$}{$Q.\mathrm{Enqueue}(s)$}\;
					}
				}
			}
			
			\Else{
				
				$P \gets P \cup \{h\}$\;
			}
		}
		\Return{$\bot$}\;
		\label{algo1}
		
		
	\end{minipage}

		\SetKwInput{KwData}{Input}

		\setcounter{AlgoLine}{0}
		\textbf{	Algorithm 2: }
		
		\KwData{$D = \{(I_1,O_1),\dots,(I_n,O_n)\}, k$}
		\KwResult{The formula \(\phi_\alpha(x,y)\) or \(\bot\)}
		
		\(\textsf{Retain} \gets \textsf{BUFIA-AC}(D, \alpha(x,y), k)\)\;
		
		\(\textsf{Add} \gets \textsf{BUFIA-AC}(D, \neg\alpha(x,y), k)\)
		
		\If{\(\textsf{Retain} =\bot \;\lor\; \textsf{Add} =\bot\)}{
			\Return{\(\bot\)} }
		\Else{
			\Return{\(\textsf{Retain} \lor \textsf{Add}\)}}
		\label{algo2}
		
		\vspace{1.2em}
		\textbf{	Algorithm 3:}
		
		\KwData{$h, I, O, h_0$}
		\KwResult{True or False}
		\If{\(h_0\)=\(\alpha(x,y)\)}{
			\If{\(h\sqsubseteq I\)}{
				\If{\(\forall (i,j) \in D(I,O)\)\\
					\Indp\([\exists r,c: r(x)=i, c(y)=j]\)}{
					\Return{True}}
		}}
		\Else{\If{\(h\not\sqsubseteq I\)}{
				\If{\(\forall (i,j) \in R(I,O)\)\\
					\Indp\([\exists r,c: r(x)=i, c(y)=j]\)}{
					\Return{True}}
		}}
		\Return{False}
		\label{algo3}
